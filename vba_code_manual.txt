Public Sub 설정()
   
    '================조건  설정================
    보종코드 = Cells(4 + i1, 2)
    갱신여부 = Cells(4 + i2, 3)
   
    If i2 = 2 Then
       보험기간 = Cells(4 + i3, 17)
    Else
       보험기간 = Cells(4 + i3, 4)
    End If
   
    제2보험기간 = Cells(4 + i4, 5)
    납입기간 = Cells(4 + i5, 6)
    납입주기 = Cells(4 + i6, 7)
    최초연령 = Cells(4 + i7, 8)
    
    If i2 = 2 Then
       주피연령 = Cells(4 + i8, 18)
    Else
       주피연령 = Cells(4 + i8, 9)
    End If

    종피연령 = Cells(4 + i9, 10)
    종피성별 = Cells(4 + i10, 11)
    표책구분코드 = Cells(4 + i11, 12)
    생사구분코드 = Cells(4 + i12, 13)
   
   
    '================보기설정================              '년만기 : 01 / 세만기 : 02
    n2 = 0 '임시 1보기2보기구분상품은 추후 업데이트
    제2보험기간구분코드 = "00" '임시 1보기2보기구분상품은 추후 업데이트
    
    If 보험기간 = 999 Then
        n = 115 - 주피연령               '[NL통합] 115 > 112 (검증은 112세까지, 준비금테이블은 115세까지 올릴것(수정을 다못했습니다.ㅠㅠ)
        보험기간구분코드 = "99"
    ElseIf 보험기간 >= 30 Then
        n = 보험기간 - 주피연령
        보험기간구분코드 = "02"
    Else
        n = 보험기간
        보험기간구분코드 = "01"
    End If
    
    '================납기설정================              '년납 : 01 / 세납 : 02
    
    If 보험기간 = 30 And 납입기간 >= 30 And 납입기간 <> 99 Then
       m = 납입기간 - 주피연령
       납기 = 납입기간
       납입기간구분코드 = "02"
    ElseIf 납입기간 = 99 Then
       m = n
       납기 = 보험기간
       납입기간구분코드 = 보험기간구분코드
    Else
       m = 납입기간
       납기 = 납입기간
       If 납입기간 = 0 Then                               'NL일시납 납기구분코드 0 (※납입기간 = 0, 납입주기 = 99)
          납입기간구분코드 = "00"
       Else
          납입기간구분코드 = "01"
       End If
    End If
    
    '================납방설정================             '월납 : 01 / 3월납 : 03 / 6월납 : 06 / 연납 : 12 / 일시납 : 99
    
    If 납입주기 = 99 Then
        납입주기구분코드 = "99"
    Else
        납입주기구분코드 = WorksheetFunction.Text(12 / 납입주기, "00")
    End If
    
End Sub

Public Sub 제한()

      If m > n _
      Or n <= 0 _
      Or m < 0 _
      Or (납입기간 = 0 And 납입주기 <> 99) _
      Or (납입기간 <> 0 And 납입주기 = 99) _
      Or 납입기간 <> 99 And (보험기간 < 45 And 납입기간 > 45) _
      Or (Cells(1, 23) = 1 And 보험기간 = 납입기간) _
      Or 주피연령 + n > Cells(28, 23) _
      Or 주피연령 + m > Cells(29, 23) _
      Or (보험기간 >= 45) And (주피연령 > (보험기간 - Cells(30, 23))) _
      Or (납입기간 >= 45) And (납입기간 <> 99) And (주피연령 > (납입기간 - Cells(31, 23))) _
      Or Cells(3, 8) > 1 And Cells(5, 8) <> 0 And (갱신여부 = "N") And (최초연령 <> 주피연령) _
      Or Cells(3, 8) > 1 And Cells(5, 8) <> 0 And (갱신여부 = "Y") And (주피연령 >= Cells(5, 9) + Cells(5, 4)) And (최초연령 < Cells(5, 9)) _
      Or Cells(3, 8) > 1 And Cells(5, 8) <> 0 And (갱신여부 = "Y") And ((주피연령 - 최초연령) / Cells(5, 4) - Int((주피연령 - 최초연령) / Cells(5, 4)) <> 0) _
      Or Cells(3, 8) > 1 And Cells(5, 8) <> 0 And (갱신여부 = "Y") And (최초연령 >= 주피연령) _
      Or 보험기간 = 30 And (납입기간 = 20 Or 납입기간 = 99 Or 납입기간 = 1) _
      Or 보험기간 = 100 And (납입기간 = 5 Or 납입기간 = 99 Or 납입기간 = 1) _
      Or Cells(34, 23) = 0 And (보험기간 = 1) _
      Or Cells(34, 23) = 1 And (보험기간 = 30 Or 보험기간 = 100) _
      Or Cells(36, 23) = 1 And (보험기간 = 30) _
      Or Cells(34, 23) = 1 And (주피연령 > 0) _
      Or Cells(33, 23) = 1 And (주피연령 > 0) _
      Or Cells(36, 23) = 1 And 보험기간 = 납입기간 And Cells(32, 23) = 1 Then '[NL통합] 납입후환급률연동형은 보험기간이 보험료납입기간을 초과하여야함/ 80세만기 80세납삭제)
         제한설정 = 1
      Else
         제한설정 = 0
      End If

End Sub
