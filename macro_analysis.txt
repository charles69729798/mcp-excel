Sub S1Check()

ii = 0

  xx = Worksheets("S산출").Cells(s1 + i, "E") '연령(주피)

Public Sub 설정()
   
    '================조건  설정================
    보종코드 = Cells(4 + i1, 2)
    갱신여부 = Cells(4 + i2, 3)

       보험기간 = Cells(4 + i3, 17)
    Else
       보험기간 = Cells(4 + i3, 4)
    End If
   
    제2보험기간 = Cells(4 + i4, 5)
    납입기간 = Cells(4 + i5, 6)
    납입주기 = Cells(4 + i6, 7)
    최초연령 = Cells(4 + i7, 8)
    
    If i2 = 2 Then
       주피연령 = Cells(4 + i8, 18)
    Else
       주피연령 = Cells(4 + i8, 9)
    End If

    종피연령 = Cells(4 + i9, 10)
    종피성별 = Cells(4 + i10, 11)
    표책구분코드 = Cells(4 + i11, 12)
    생사구분코드 = Cells(4 + i12, 13)
   
   
    '================보기설정================              '년만기 : 01 / 세만기 : 02
    n2 = 0 '임시 1보기2보기구분상품은 추후 업데이트
    제2보험기간구분코드 = "00" '임시 1보기2보기구분상품은 추후 업데이트
    
    If 보험기간 = 999 Then
        n = 115 - 주피연령               '[NL통합] 115 > 112 (검증은 112세까지, 준비금테이블은 115세까지 올릴것(수정을 다못했습니다.ㅠㅠ)
        보험기간구분코드 = "99"
    ElseIf 보험기간 >= 30 Then
        n = 보험기간 - 주피연령
        보험기간구분코드 = "02"
    Else
        n = 보험기간
        보험기간구분코드 = "01"
    End If
    
    '================납기설정================              '년납 : 01 / 세납 : 02
    
    If 보험기간 = 30 And 납입기간 >= 30 And 납입기간 <> 99 Then
       m = 납입기간 - 주피연령
       납기 = 납입기간
       납입기간구분코드 = "02"
    ElseIf 납입기간 = 99 Then
       m = n
       납기 = 보험기간
       납입기간구분코드 = 보험기간구분코드
    Else
